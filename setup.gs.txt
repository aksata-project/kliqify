/**
 * Gets the primary Users sheet.
 */
function getUsersSheet() {
  const ss = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID);
  
  // STRICTLY look for the configured sheet first
  let sheet = ss.getSheetByName(CONFIG.SHEET_USERS);
  
  // If not found, create it immediately to enforce the new schema
  if (!sheet) {
    console.log(`Sheet '${CONFIG.SHEET_USERS}' not found. Creating new one.`);
    sheet = ss.insertSheet(CONFIG.SHEET_USERS);
    SpreadsheetApp.flush();
  }
  
  // Always ensure headers on the returned sheet
  ensureHeaders(sheet);
  
  return sheet;
}

/**
 * Returns all sheets that might contain user data.
 */
function getAllUserSheets() {
  const ss = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID);
  const names = [CONFIG.SHEET_MASTER, CONFIG.SHEET_USERS, ...CONFIG.SHEET_FALLBACKS];
  const sheets = [];
  const seen = new Set();
  
  for (const name of names) {
    if (seen.has(name)) continue;
    const s = ss.getSheetByName(name);
    if (s) {
      ensureHeaders(s); // Force injecting the Links_Claimed column if missing
      sheets.push(s);
      seen.add(name);
    }
  }
  return sheets;
}

/**
 * Ensures all required columns exist in the sheet.
 * Appends missing columns to the end of Row 1.
 */
function ensureHeaders(sheet) {
  let lastCol = sheet.getLastColumn();
  let headers = [];
  
  if (lastCol > 0) {
    try {
      headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
    } catch (e) {
      console.warn("Error reading headers, might be empty", e);
      lastCol = 0;
    }
  }
  
  const missing = [];
  
  for (const [key, aliases] of Object.entries(CONFIG.COLS_USERS)) {
    if (findColumnIndex(headers, aliases) === -1) {
      missing.push(aliases[0]); // Add the canonical name
    }
  }
  
  if (missing.length > 0) {
    console.log(`Repairing headers for '${sheet.getName()}'. Adding: ${missing.join(', ')}`);
    const currentLastCol = sheet.getLastColumn();
    sheet.getRange(1, currentLastCol + 1, 1, missing.length).setValues([missing]);
    SpreadsheetApp.flush(); // Critical to ensure visibility for next operations
  }
}

/**
 * Helper to find column index case-insensitively.
 * @param {Array} headers - Row 1 of the sheet
 * @param {String} columnName - The column name to find (e.g. "Username")
 * @return {Integer} - 0-based index or -1 if not found
 */
/**
 * Helper to find column index case-insensitively with alias support.
 * @param {Array} headers - Row 1 of the sheet
 * @param {String|Array} columnName - The column name(s) to find
 * @return {Integer} - 0-based index or -1 if not found
 */
function findColumnIndex(headers, columnName) {
  if (!headers || !columnName) return -1;
  
  // Convert search terms to array of lowercased strings
  const searchTerms = (Array.isArray(columnName) ? columnName : [columnName])
    .map(s => String(s).toLowerCase().trim());
  
  for (let i = 0; i < headers.length; i++) {
    const header = String(headers[i]).toLowerCase().trim();
    if (searchTerms.includes(header)) {
      return i;
    }
  }
  return -1;
}

/**
 * Get all users data as an array of objects
 */
function getAllUsers() {
  const sheet = getUsersSheet();
  const data = sheet.getDataRange().getValues();
  if (data.length < 2) return [];
  
  const headers = data[0];
  const users = [];
  
  // Map columns dynamically
  const colMap = {};
  for (const [key, val] of Object.entries(CONFIG.COLS_USERS)) {
    colMap[key] = findColumnIndex(headers, val);
  }
  
  // Determine key columns
  const userIdx = colMap.USERNAME !== -1 ? colMap.USERNAME : findColumnIndex(headers, 'username');
  
  if (userIdx === -1) return []; // Cannot parse without username
  
  return data.slice(1);
}

}

/**
 * UTILITY: FORCE INJECT CLAIMED COLUMN
 * Run this if MANUAL_SETUP fails to show the column.
 * This directly forces the column 'Claimed' at the end of the sheet.
 */
function FORCE_INJECT_CLAIMED_COLUMN() {
  const ss = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID);
  const sheets = [ss.getSheetByName(CONFIG.SHEET_USERS), ss.getSheetByName('Users')].filter(Boolean);
  
  sheets.forEach(sheet => {
    const lastCol = sheet.getLastColumn();
    const headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
    if (!headers.includes('Claimed') && !headers.includes('Links_Claimed')) {
      sheet.getRange(1, lastCol + 1).setValue('Claimed');
      console.log(`✅ Forced 'Claimed' column into sheet: ${sheet.getName()} at column ${lastCol + 1}`);
    } else {
      console.log(`⚠️ Column 'Claimed' already exists in sheet: ${sheet.getName()}`);
    }
  });
  SpreadsheetApp.flush();
}
